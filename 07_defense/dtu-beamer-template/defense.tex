\section{Framework and method}
% =====================================================================

\begin{frame}{Four theories, one analytical spine}
    \begin{itemize}
        \item \textbf{Scott} (2014) — \emph{pillars}: where in an institution a claim is anchored.
        \item \textbf{DiMaggio \& Powell} (1983) — \emph{mechanism}: how claims travel between organisations.
        \item \textbf{Thornton et al.} (2012) — \emph{content}: which logic a claim expresses.
        \item \textbf{Suchman} (1995) — \emph{outcome}: how legitimacy is secured.
    \end{itemize}
    \vspace{3mm}
    Removing any one breaks a specific finding.
\end{frame}

\begin{frame}{Method — three-pass documentary analysis}
    \textbf{Unit of analysis:} discursive construction \emph{inside the text}. Not authorial intent.

    \vspace{3mm}
    \textbf{Pipeline.}
    \begin{itemize}
        \item Pass 1 — \textbf{deductive}. 180 entries against a-priori codebooks.
        \item Pass 2 — \textbf{inductive}. 29 flags resolved into 6 patterns, 7 coding rules.
        \item Pass 3 — writing.
    \end{itemize}

    \vspace{2mm}
    \textbf{GenAI-assisted with three safeguards.} A-priori codebooks. Page provenance. Manual colour-coded cross-reference.

    \vspace{2mm}
    \textbf{Single coder.} Limitation — mitigated, not eliminated.
\end{frame}

\begin{frame}{A moment the analysis had to correct itself}
    \textbf{Initial reading.} Corpus explicitly benchmarks against DARPA, Temasek, Chinese GGF.

    \vspace{1mm}
    \textbf{Corpus review.} No named comparator. Strategic Goal 4 says \enquote{USA and Asia}.

    \vspace{2mm}
    \textbf{Source of the drift.} Parallel literature (Mazzucato, Karo \& Kattel, Bonvillian) leaked into how I read the primary text.

    \vspace{3mm}
    \textbf{The reading was not wrong.} DiMaggio \& Powell define mimetic isomorphism as \emph{structural resemblance}, not textual acknowledgement. Finding survived; framing corrected. Audit trail: synthesis §0.5.

    \vspace{2mm}
    \textbf{The safeguard worked.} Provenance check caught the drift.
\end{frame}

\section{Findings}
% =====================================================================

\begin{frame}{Headline — logic succession across TRL}
    \begin{block}{}
        Across the TRL axis, WP 2026 organises as \textbf{Science $\to$ Market $\to$ State} — and the same succession is reproduced \emph{inside each Challenge chapter}.
    \end{block}
    \vspace{3mm}
    Two scales at once.
    \begin{itemize}
        \item Aggregate: across the TRL axis.
        \item Structural: inside every thematic block. \textbf{Pattern D}.
    \end{itemize}
    \vspace{2mm}
    \footnotesize The kind of pattern you only see once you have coded provenance.
\end{frame}

\begin{frame}{Pattern A — Tension-Dissolution}
    Competing logics presented as complementary. Rule 1 keeps the code honest: tension must be traceable in the text.

    \vspace{3mm}
    \textbf{Instrument-level extension.}
    \begin{itemize}
        \item \emph{Register:} market vocabulary — Venture Debt, equity, deal flow.
        \item \emph{Organising:} state — Programme Manager authority, Commission conditionalities, sovereign access.
    \end{itemize}
    \vspace{2mm}
    Rule 7 codifies the register-vs-organising check.

    \vspace{3mm}
    \footnotesize Pre-empt: state aid is a legal category (Art. 107). Pattern A is an institutional-legitimation move — different register.
\end{frame}

\begin{frame}{Pattern E — STEP Circular Market Validation}
    \begin{center}
        \small state capital $\to$ co-investment $\to$ \emph{treated as market signal} $\to$ justifies more state capital
    \end{center}
    \vspace{3mm}
    The circle closes on itself. \textbf{Rule 2} codifies the STEP catalytic co-coding.

    \vspace{3mm}
    Adjacent to Gabor (WSC critique); Mertens \& Thiemann (de-risking). Arrived at through document-internal coding, not applied from outside.

    \vspace{3mm}
    \footnotesize The pattern surfaces a rhetorical move the document works to keep invisible.
\end{frame}

\begin{frame}{The Sovereignty Seal — three coded functions}
    \begin{itemize}
        \item Unlocks \textbf{InvestEU Venture Debt} — entries [152], [161].
        \item Unlocks \textbf{Business Acceleration Services}.
        \item Awardable on a \textbf{No-Go decision} — entry [166].
    \end{itemize}
    \vspace{4mm}
    \textbf{Decoupling from the investment decision} — the state certifies strategic value independently of committing capital.

    \vspace{3mm}
    Field-bridging: portable across public EU programmes and private investors. Synthesis §2.8.

    \vspace{3mm}
    \footnotesize Findings gives the artefact. Discussion asks what it is doing.
\end{frame}

\begin{frame}{Brief mentions — Patterns B, C, F, and a null}
    \begin{itemize}
        \item \textbf{Pattern B} — Coercive Governance Cluster. Programme Manager authority gradient. Rule 3.
        \item \textbf{Pattern C} — State Social Equity Template. Standardised DEI text, state-logic anchored. Rule 4.
        \item \textbf{Pattern F} — TRL as Cross-Pillar Legitimation Artefact. Regulative + cognitive + mimetic triple-load.
    \end{itemize}
    \vspace{3mm}
    \textbf{Null result — Sustainability template not confirmed.} Four preserved entry mechanisms; analytical contrast with Pattern C.
\end{frame}

\section{Discussion}
% =====================================================================

\begin{frame}{STEP as incomplete developmental-state mimicry}
    \begin{center}
    \begin{tabular}{@{}lcc@{}}
        \toprule
                                    & \textbf{Supply}  & \textbf{Demand}                       \\ \midrule
        \textbf{DARPA}              & missions funded  & DoD off-take (largest single customer)\\
        \textbf{Chinese GGF}        & guidance capital & SOE off-take                          \\
        \textbf{STEP} (WP 2026)     & catalytic capital& \textbf{— absent —}                   \\ \bottomrule
    \end{tabular}
    \end{center}
    \vspace{3mm}
    Not an oversight. A structural feature of doing developmental-state work under \textbf{Article 107} state-aid constraints.

    \vspace{2mm}
    \footnotesize Structural comparison per DP\&P's original meaning. Read against Mazzucato, Karo \& Kattel, Bonvillian.
\end{frame}

\begin{frame}{The Seal as candidate demand-side instrument}
    \textbf{Corpus chain:} Seal $\to$ BAS $\to$ Ch VII $\to$ VII.5 \emph{Strategic Access to Innovation Procurement}.

    \vspace{2mm}
    \footnotesize (VII.5 body outside sliced primary corpus — this is \textbf{a reflection register, not a finding}.)

    \vspace{3mm}
    Seal is \textbf{ambiguous by design}. Not \enquote{for procurement} — \enquote{strategically important}. That ambiguity lets it work across two channels at once:
    \begin{itemize}
        \item Investor legitimacy
        \item Procurement discretion — under existing procurement law, without triggering Art. 107 review
    \end{itemize}

    \vspace{2mm}
    \footnotesize Adjacencies: Mazzucato, Karo \& Kattel, Draghi, Letta.
\end{frame}

\begin{frame}{Practice implications}
    \textbf{Policy officer (next instrument).} Demand-side statecraft is the under-designed side. Seal is a candidate design pattern.

    \vspace{2mm}
    \textbf{EIC evaluator.} Seal is not a consolation prize. Operational credential. Separable from the investment decision — as entry [166] already positions it.

    \vspace{2mm}
    \textbf{Would-be STEP applicant.} Seal has downstream value — InvestEU + BAS gateway + procurement discretion signal — independent of whether STEP funds the round.
\end{frame}

\section{Limitations}
% =====================================================================

\begin{frame}{Limitations}
    \begin{itemize}
        \item \textbf{Single-document primary corpus.} Unit is discursive construction \emph{inside} this instrument.
        \item \textbf{Single-coder interpretive analysis.} Mitigated by codified rules, provenance, cross-reference. Not eliminated.
        \item \textbf{Point-in-time.} WP 2026 will be superseded. Patterns speak to how the document does its institutional work — transferable in mechanism if not in specifics.
        \item \textbf{No implementation data.} Studied the text, not the deployment.
    \end{itemize}
\end{frame}
